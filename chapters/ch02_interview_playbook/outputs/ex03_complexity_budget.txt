  n =            10: O(n!)
  n =            20: O(2^n)
  n =           500: O(n^2)
  n =         3,000: O(n^2)
  n =       100,000: O(n log n)
  n =     1,000,000: O(n)
  n = 1,000,000,000: O(log n)
